# Enable Jekyll for GitHub Pages blog auto-listing

# 1. Create a _config.yml file (if not present)
# This file tells GitHub Pages to use Jekyll and sets up basic config.

# 2. Move your blog posts to a _posts/ folder and rename them as:
#   YYYY-MM-DD-title-of-post.md (or .html)
#   Example: 2025-08-31-spring-transaction-management.html

# 3. Create an index.html (or blog.html) with Jekyll/Liquid code to auto-list posts.

# 4. Commit and push to GitHub Pages branch (gh-pages or main)

# --- _config.yml ---

# Minimal config for Jekyll
markdown: kramdown
permalink: pretty
exclude: ["node_modules", "gulpfile.js", "package.json", "package-lock.json"]

# --- Code Blocks with Copy Button ---
# Add this to your blog's <head> (in _layouts/default.html or blog/index.html):
#
# <style>
# pre code {
#   display: block;
#   background: #222;
#   color: #f8f8f2;
#   padding: 1em;
#   border-radius: 8px;
#   font-size: 1.05em;
#   overflow-x: auto;
#   position: relative;
# }
# .copy-btn {
#   position: absolute;
#   top: 8px;
#   right: 12px;
#   background: #BF5700;
#   color: #fff;
#   border: none;
#   border-radius: 4px;
#   padding: 0.3em 0.8em;
#   cursor: pointer;
#   font-size: 0.95em;
#   opacity: 0.85;
#   transition: opacity 0.2s;
#   z-index: 2;
# }
# .copy-btn:hover { opacity: 1; }
# </style>
#
# Add this before </body> (in _layouts/default.html or blog/index.html):
#
# <script>
# document.querySelectorAll('pre code').forEach(function(codeBlock) {
#   var btn = document.createElement('button');
#   btn.className = 'copy-btn';
#   btn.type = 'button';
#   btn.innerText = 'Copy';
#   btn.onclick = function() {
#     navigator.clipboard.writeText(codeBlock.innerText);
#     btn.innerText = 'Copied!';
#     setTimeout(function(){ btn.innerText = 'Copy'; }, 1200);
#   };
#   codeBlock.parentNode.style.position = 'relative';
#   codeBlock.parentNode.appendChild(btn);
# });
# </script>
#
# Now, all <pre><code> blocks will have a styled look and a one-click copy button.

# To create multiple Angular posts:
# 1. Copy the template above for each new post.
# 2. Change the `title`, `date`, and content for each post.
# 3. Use the `categories: angular` line to group all Angular posts together.
# 4. Example filenames:
#    _posts/2025-09-23-getting-started-with-angular.md
#    _posts/2025-09-24-angular-components-basics.md
#    _posts/2025-09-25-angular-routing-tutorial.md
